# Dependencies for the self-verification gate.
#
# Mirrors the test dependencies HAIR declares for itself in pyproject.toml.
# The gate imports HAIR's own modules and uses them as the independent
# witness, so it should run with what HAIR's own tests run with, no more and
# no less. When HAIR's ranges move, these move with them.
#
# Python 3.13 is the floor (setup.sh, ci.yml). HAIR declares
# requires-python >=3.13, and below it the upstream decoders cannot be
# installed at all.

# cryptography verifies ed25519 fitting signatures. Without it a signed
# fitting cannot be checked, and the gate reports invalid rather than
# vouching blind, which fails the run.
cryptography>=42.0

# PyYAML is what HAIR's field readers load their field maps with, and the
# comb imports the field readers. Without it the comb cannot be imported at
# all. Missing from this file from 2026-08-24, when HAIR 0.12.0 introduced
# the dependency, until 2026-09-22, and every weekly CI run in between died
# on the import before it gated a single wig.
pyyaml>=6.0

# infrared-protocols carries the decoders HAIR has no local copy of, NEC
# above all. Without it, every NEC wig reads as "does not decode", which is
# a false statement about a good wig, so the gate refuses to start rather
# than run without it (verify_wig.Hair).
#
# The range is HAIR's (0.17.2's pyproject). It resolves to 5.8.1 on Python
# 3.13 and the newest 9.x on 3.14, because 8.x and later declare
# Requires-Python >=3.14. CI runs both legs, so both upstream versions are
# exercised against the same wigs. Home Assistant itself has moved on to
# 10.x; the gate passed every check against 10.2.0 on 2026-10-13, and this
# follows HAIR when HAIR lifts its cap.
infrared-protocols>=5.6,<10.0
